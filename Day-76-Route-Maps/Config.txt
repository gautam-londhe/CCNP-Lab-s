R1 (AS65001)
============

conf t

hostname R1

interface g0/0
ip address 10.12.12.2 255.255.255.252
no shutdown

interface g0/1
ip address 10.13.13.1 255.255.255.252
no shutdown

ip prefix-list CLOUD seq 10 permit 100.100.100.0/24
ip prefix-list CLOUD seq 20 permit 101.101.101.0/24

route-map CLOUD-POLICY permit 10
match ip address prefix-list CLOUD
set local-preference 500

router bgp 65001

neighbor 10.12.12.1 remote-as 65002
neighbor 10.13.13.2 remote-as 65001

neighbor 10.12.12.1 route-map CLOUD-POLICY in

end
