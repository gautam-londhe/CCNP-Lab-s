R1 (Enterprise Edge)

hostname R1

interface GigabitEthernet0/0
ip address 10.12.12.2 255.255.255.252
no shutdown

interface GigabitEthernet0/1
ip address 10.13.13.1 255.255.255.252
no shutdown

interface GigabitEthernet0/2
ip address 10.23.23.1 255.255.255.252
no shutdown

router ospf 1
network 10.12.12.0 0.0.0.3 area 0
network 10.13.13.0 0.0.0.3 area 0
network 10.23.23.0 0.0.0.3 area 0

router bgp 65001
bgp log-neighbor-changes
neighbor 10.12.12.1 remote-as 65002

network 10.1.1.1 mask 255.255.255.255
network 10.2.2.1 mask 255.255.255.255
network 10.3.3.1 mask 255.255.255.255
network 10.4.4.1 mask 255.255.255.255

neighbor 10.12.12.1 prefix-list BRANCHES out

ip prefix-list BRANCHES seq 10 permit 10.1.1.1/32
ip prefix-list BRANCHES seq 20 permit 10.2.2.1/32
